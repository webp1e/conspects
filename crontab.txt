cron - планировщик задач

crontab -e  # редактировать задачи
crontab -l  # посмотреть задачи

* * * * * команда
│ │ │ │ │
│ │ │ │ └── день недели (0-7)
│ │ │ └──── месяц (1-12)
│ │ └────── день месяца (1-31)
│ └──────── час (0-23)
└────────── минута (0-59)

каждую минуту
* * * * * /home/user/script.sh

каждый день в 3 ночи
0 3 * * * /home/user/script.sh

каждый понедельник в 9 утра
0 9 * * 1 /home/user/script.sh
