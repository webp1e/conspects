базовые команды
nmap -sV 192.168.0.0/24     скан локальной сети
nmap -p 1-1000 192.168.1.1  скан диапазон портов айпи
nmap 192.168.1.1            скан хоста
nmap -p- 192.168.1.1        скан всех портов хоста
nmap -sV 192.168.1.1        скан с определением версии сервисов
nmap -O 192.168.1.1         определить ос 
nmap -A 192.168.1.1         определить всё сразу
sudo nmap -sS 192.168.1.1   SYN скан, быстрее но нужен sudo
nmap localhost              скан себя
